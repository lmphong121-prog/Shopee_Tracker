Đồng bộ dữ liệu Đám mây (Đa thiết bị)

Tự động đồng bộ đơn mua hàng, đơn Shopee, tồn kho & sản phẩm DIY giữa máy tính và điện thoại

Chưa kết nối đám mây. Nhập URL Google Apps Script hoặc Firebase để bắt đầu.
Hướng dẫn tạo máy chủ đồng bộ miễn phí bằng Google Sheets (1 phút)
  1. Tạo 1 file Google Sheets trống trên Google Drive của bạn.
  2. Trên thanh menu chọn Tiện ích mở rộng (Extensions) → Apps Script.
  3. Xóa mã cũ, bấm nút Sao chép mã Apps Script bên dưới rồi dán vào (đổi SECRET_PIN nếu muốn đặt mật khẩu riêng).
  4. Bấm Triển khai (Deploy) → Tùy chọn triển khai mới → Ứng dụng web (Web app): chọn Thực thi dưới dạng: Tôi (Me) và Ai có quyền truy cập: Bất kỳ ai (Anyone) → bấm Triển khai.
  5. Sao chép đường link https://script.google.com/macros/s/.../exec dán vào ô URL phía trên (và dán cùng URL + mã PIN trên điện thoại để đồng bộ tự động).
Mã Google Apps Script: 
const SECRET_PIN = ""; // Đổi thành mã PIN của bạn (VD: "123456") nếu muốn bảo mật

function getStoreSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName("ShopeeSyncData") || ss.insertSheet("ShopeeSyncData");
}

function jsonOut(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  const token = (e && e.parameter && e.parameter.token) || "";
  if (SECRET_PIN && token !== SECRET_PIN) return jsonOut({ ok: false, error: "Sai mã PIN" });
  const sheet = getStoreSheet();
  const raw = sheet.getRange("A1").getValue();
  const updatedAt = sheet.getRange("B1").getValue();
  if (!raw) return jsonOut({ ok: true, data: null, updatedAt: "" });
  return jsonOut({ ok: true, data: JSON.parse(raw), updatedAt: String(updatedAt || "") });
}

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    if (SECRET_PIN && (body.token || "") !== SECRET_PIN) return jsonOut({ ok: false, error: "Sai mã PIN" });
    const sheet = getStoreSheet();
    const now = new Date().toISOString();
    sheet.getRange("A1").setValue(JSON.stringify(body.data));
    sheet.getRange("B1").setValue(now);
    return jsonOut({ ok: true, updatedAt: now });
  } catch (err) {
    return jsonOut({ ok: false, error: String(err) });
  }
}
